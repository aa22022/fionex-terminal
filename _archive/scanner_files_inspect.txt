    for i in (p + 1):n

function calculate_rsi(prices::Vector{Float64}, period::Int=14)::Float64
    n = length(prices)
    p = min(period, n - 1)
    p <= 2 && return 50.0

    losses = 0.0
    for i in (n - p + 1):n
        diff = prices[i] - prices[i-1]
        if diff > 0
            gains += diff
        else
            losses -= diff
        end
    end
    
    avg_gain = gains / p
    avg_loss = losses / p
    avg_loss == 0.0 && return 100.0
    
    rs = avg_gain / avg_loss
    return 100.0 - (100.0 / (1.0 + rs))
end
