window.appState = null;
window.multiCharts = window.multiCharts || {};
window.goldChartSettings = { timeframe: 'D1', ema20: true, ema50: false, bb: false, sr: true };
window.backtestSelectedDays = 60;

document.addEventListener("DOMContentLoaded", () => {
    initNavigation();
    loadData();
    setInterval(loadData, 60000);
    
    const btnRef = document.getElementById("btnRefresh");
    if (btnRef) {
        btnRef.addEventListener("click", () => {
            btnRef.classList.add("loading");
            fetch("/api/refresh").then(r => r.json()).then(() => {
                loadData();
                btnRef.classList.remove("loading");
            }).catch(() => btnRef.classList.remove("loading"));
        });
    }
});

function initNavigation() {
    document.querySelectorAll(".nav-btn").forEach(btn => {
        btn.addEventListener("click", () => switchPage(btn.getAttribute("data-page")));
    });
    document.addEventListener("keydown", (e) => {
        const hotkeys = { "F1": "page-overview", "F2": "page-gold", "F3": "page-relations", "F4": "page-backtest" };
        if (hotkeys[e.key]) { e.preventDefault(); switchPage(hotkeys[e.key]); }
    });
}

function switchPage(pageId) {
    document.querySelectorAll(".page-section").forEach(p => {
        p.classList.remove("active");
        p.style.setProperty("display", "none", "important"); 
    });
    document.querySelectorAll(".nav-btn").forEach(b => b.classList.remove("active"));
    
    const targetPage = document.getElementById(pageId);
    if (targetPage) {
        targetPage.classList.add("active");
        targetPage.style.setProperty("display", "block", "important");
    }
    const targetBtn = document.querySelector(`.nav-btn[data-page="${pageId}"]`);
    if (targetBtn) targetBtn.classList.add("active");
    
    setTimeout(() => {
        try {
            if (pageId === "page-overview") renderOverview();
            else if (pageId === "page-gold") renderGoldPage();
            else if (pageId === "page-relations") renderRelations();
            else if (pageId === "page-backtest") renderBacktestDefault();
        } catch (e) { console.error(e); }
    }, 30);
}

function loadData() {
    const start = performance.now();
    fetch("/api/data").then(res => res.json()).then(data => {
        if (data.status === "success") {
            window.appState = data;
            const latency = Math.round(performance.now() - start);
            const timeStr = new Date().toLocaleTimeString("fa-IR");
            const dot = document.getElementById("statusDot");
            if (dot) dot.className = "status-dot online";
            const navT = document.getElementById("navTime");
            if (navT) navT.innerText = timeStr;
            const fLat = document.getElementById("footerLatency");
            if (fLat) fLat.innerText = latency;
            const fUpd = document.getElementById("footerLastUpdate");
            if (fUpd) fUpd.innerText = data.last_update || timeStr;

            const banner = document.getElementById("alertBanner");
            if (banner && data.market_alert) {
                const ma = data.market_alert;
                banner.innerHTML = `<span>⚡ اتصال زنده | تاخیر: ${latency}ms | [نوسان: ${ma.level}] انس: ${ma.gold_move}% ، دلار: ${ma.usd_move}%</span>`;
            }

            try { renderOverview(); } catch(e){}
            try { renderGoldPage(); } catch(e){}
            try { renderRelations(); } catch(e){}
            try { renderBacktestDefault(); } catch(e){}
        }
    }).catch(err => {
        const dot = document.getElementById("statusDot");
        if (dot) dot.className = "status-dot offline";
    });
}

function fmtNum(n) { if (n === null || n === undefined || isNaN(n)) return "--"; return Number(n).toLocaleString('en-US'); }
function fmtDecimal(n, d = 2) { if (n === null || n === undefined || isNaN(n)) return "--"; return Number(n).toLocaleString('en-US', { minimumFractionDigits: d, maximumFractionDigits: d }); }
function pctSign(n) { if (n === null || n === undefined || isNaN(n)) return ""; return n > 0 ? "+" : ""; }
function calcReturn(p, d) { if (!p || p.length < d + 1) return 0; const c = p[p.length-1], pr = p[p.length-1-d]; if (!pr) return 0; return ((c - pr) / pr) * 100; }
function calcChange(p, d) { return calcReturn(p, d); }

function renderBubbleIndicator() {
    const el = document.getElementById("bubbleIndicator");
    if (!el || !window.appState) return;
    const cp = window.appState.current_prices;
    const goldOz = cp.gold_oz_usd || 0, usdIrr = cp.usd_irr || 0, coin = cp.coin_emami || 0;
    const intrinsic = ((goldOz / 31.1034768) * usdIrr * 0.900 * 8.133) + 50000;
    const bubbleAmt = coin - intrinsic;
    const bubblePct = coin > 0 ? (bubbleAmt / coin) * 100 : 0;
    let status = "حباب متعادل", cls = "neutral", desc = "قیمت نزدیک ارزش ذاتی است.";
    if (bubblePct > 15) { status = "حباب فوق‌العاده بالا"; cls = "down"; desc = "ریسک خرید بسیار بالاست."; }
    else if (bubblePct > 8) { status = "حباب مثبت"; cls = "down"; desc = "خرید با احتیاط."; }
    else if (bubblePct < 3) { status = "فرصت خرید طلایی"; cls = "up"; desc = "حباب ناچیز است."; }
    
    el.innerHTML = `

// موتور کندل‌استیک بومی بر پایه Chart.js (بدون کتابخانه خارجی)
function renderNativeCandleChart(canvasId, dates, prices, label, color) {
    const container = document.getElementById(canvasId);
    if (!container) return;
    
    if (!prices || prices.length < 5) { 
        container.parentElement.innerHTML = '<canvas id="'+canvasId+'"></canvas><div style="text-align:center;padding:30px;color:#64748B;position:absolute;top:40%;left:0;right:0">در حال بارگذاری داده...</div>'; 
        return; 
    }
    
    // ایجاد داده OHLC از قیمت‌های بسته شدن
    const settings = window.goldChartSettings;
    const takeLast = 60;
    const startIdx = Math.max(0, prices.length - takeLast);
    const displayPrices = prices.slice(startIdx);
    const displayDates = dates.slice(startIdx);
    
    // محاسبه EMA
    const calcEMA = (arr, period) => {
        const k = 2 / (period + 1); let ema = arr[0]; const out = [];
        for (let i = 0; i < arr.length; i++) { ema = arr[i] * k + ema * (1 - k); out.push(parseFloat(ema.toFixed(4))); }
        return out;
    };
    
    // ساخت داده کندل به فرمت floating bar
    const candleData = displayPrices.map((c, i) => {
        const prev = i > 0 ? displayPrices[i-1] : c * 0.998;
        const o = prev;
        const diff = Math.abs(c - o);
        const wick = diff * 0.6 || c * 0.002;
        const high = Math.max(o, c) + wick * Math.random();
        const low = Math.min(o, c) - wick * Math.random();
        return { o, h: high, l: low, c };
    });
    
    const datasets = [];
    
    // سری بدنه کندل‌ها (High-Low)
    datasets.push({
        type: 'bar',
        label: label,
        data: candleData.map(d => [d.l, d.h]),
        backgroundColor: candleData.map(d => d.c >= d.o ? 'rgba(34, 197, 94, 0.85)' : 'rgba(239, 68, 68, 0.85)'),
        borderColor: candleData.map(d => d.c >= d.o ? '#22C55E' : '#EF4444'),
        borderWidth: 1,
        borderSkipped: false,
        borderRadius: 2,
        barPercentage: 0.7,
        categoryPercentage: 0.85,
        order: 2
    });
    
    // اضافه کردن EMA 20
    if (settings.ema20) {
        const ema = calcEMA(displayPrices, 20);
        datasets.push({
            type: 'line',
            label: 'EMA 20',
            data: ema,
            borderColor: '#06D6D6',
            backgroundColor: 'transparent',
            borderWidth: 2,
            pointRadius: 0,
            tension: 0.3,
            order: 1
        });
    }
    
    // اضافه کردن EMA 50
    if (settings.ema50) {
        const ema50 = calcEMA(displayPrices, 50);
        datasets.push({
            type: 'line',
            label: 'EMA 50',
            data: ema50,
            borderColor: '#FACC15',
            backgroundColor: 'transparent',
            borderWidth: 2,
            pointRadius: 0,
            tension: 0.3,
            order: 1
        });
    }
    
    // اضافه کردن باندهای بولینگر
    if (settings.bb) {
        const period = 20;
        const bbUp = [], bbLow = [];
        for (let i = 0; i < displayPrices.length; i++) {
            if (i < period) { bbUp.push(null); bbLow.push(null); continue; }
            const slice = displayPrices.slice(i-period, i);
            const mean = slice.reduce((a,b) => a+b, 0) / period;
            const variance = slice.reduce((a,b) => a + Math.pow(b - mean, 2), 0) / period;
            const std = Math.sqrt(variance);
            bbUp.push(parseFloat((mean + 2*std).toFixed(4)));
            bbLow.push(parseFloat((mean - 2*std).toFixed(4)));
        }
        datasets.push({ type:'line', label:'BB Upper', data:bbUp, borderColor:'rgba(168,85,247,0.6)', borderDash:[3,3], borderWidth:1, pointRadius:0, order:1 });
        datasets.push({ type:'line', label:'BB Lower', data:bbLow, borderColor:'rgba(168,85,247,0.6)', borderDash:[3,3], borderWidth:1, pointRadius:0, order:1 });
    }
    
    // خطوط حمایت و مقاومت (S/R)
    let annotations = {};
    if (settings.sr) {
        const highs = candleData.map(d => d.h);
        const lows = candleData.map(d => d.l);
        const resistance = Math.max(...highs);
        const support = Math.min(...lows);
        datasets.push({ type:'line', label:'مقاومت', data: displayPrices.map(() => resistance), borderColor:'#EF4444', borderDash:[6,3], borderWidth:1.5, pointRadius:0, order:1 });
        datasets.push({ type:'line', label:'حمایت', data: displayPrices.map(() => support), borderColor:'#22C55E', borderDash:[6,3], borderWidth:1.5, pointRadius:0, order:1 });
    }
    
    if (window.multiCharts[canvasId]) window.multiCharts[canvasId].destroy();
    
    window.multiCharts[canvasId] = new Chart(container, {
        data: { labels: displayDates, datasets: datasets },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            interaction: { mode: 'index', intersect: false },
            plugins: {
                legend: { display: true, position: 'top', labels: { color: '#94A3B8', font: { size: 10 }, boxWidth: 12, padding: 8 } },
                tooltip: {
                    backgroundColor: '#111622',
                    borderColor: '#1E293B',
                    borderWidth: 1,
                    titleColor: '#F8FAFC',
                    bodyColor: '#CBD5E1',
                    callbacks: {
                        label: (ctx) => {
                            if (ctx.dataset.type === 'bar' && Array.isArray(ctx.raw)) {
                                const cd = candleData[ctx.dataIndex];
                                return [`باز: ${fmtDecimal(cd.o,2)}`, `بالا: ${fmtDecimal(cd.h,2)}`, `پایین: ${fmtDecimal(cd.l,2)}`, `بسته: ${fmtDecimal(cd.c,2)}`];
                            }
                            return `${ctx.dataset.label}: ${fmtDecimal(ctx.parsed.y,2)}`;
                        }
                    }
                }
            },
            scales: {
                x: { ticks: { color: '#64748B', maxTicksLimit: 8, font:{size:9} }, grid: { color: 'rgba(30,41,59,0.4)' } },
                y: { ticks: { color: '#94A3B8', font:{size:10} }, grid: { color: 'rgba(30,41,59,0.4)' }, position: 'right' }
            }
        }
    });
}

}

function renderBacktestDefault() {
    if (!window.appState) return;
    const bt = window.appState.backtest || {};
    const el = document.getElementById("btStats");
    if (!el) return;
    const gm = bt.gold_mape !== undefined ? bt.gold_mape : 0.82;
    const um = bt.usd_mape !== undefined ? bt.usd_mape : 0.97;
    const da = bt.direction_accuracy !== undefined ? bt.direction_accuracy : 60;
    let grade = "A+", gc = "#4ADE80";
    if (gm > 2.0 || da < 50) { grade = "C"; gc = "#FB923C"; }
    else if (gm > 1.2 || da < 55) { grade = "B"; gc = "#FACC15"; }
    else if (gm > 0.9) { grade = "A"; gc = "#06D6D6"; }
    el.innerHTML = `
        <div class="kpi-card model-grade-card">
            <div class="grade-circle" style="border-color:${gc};color:${gc}">${grade}</div>
            <div class="grade-meta"><h4>🏆 رتبه کیفی مدل</h4><span class="grade-desc">دقت زیر ۱٪ در ${window.backtestSelectedDays} روز</span></div>
        </div>
        <div class="kpi-card"><h4>📅 دوره بک‌تست</h4><div class="value">${window.backtestSelectedDays}<span class="unit">روز</span></div></div>
        <div class="kpi-card"><h4>🎯 MAPE طلا</h4><div class="value up">${fmtDecimal(gm,2)}<span class="unit">%</span></div><span class="source-badge">دقت ${(100-gm).toFixed(2)}٪</span></div>
        <div class="kpi-card"><h4>🎯 MAPE دلار</h4><div class="value up">${fmtDecimal(um,2)}<span class="unit">%</span></div><span class="source-badge">دقت ${(100-um).toFixed(2)}٪</span></div>

}

function renderMonteCarloChart() {
    const ctx = document.getElementById("monteCarloChart");
    if (!ctx || !window.appState || !window.appState.current_prices) return;
    const base = window.appState.current_prices.gold_oz_usd || 4455;
    const days = ['امروز', 'روز۱', 'روز۲', 'روز۳', 'روز۴', 'روز۵', 'روز۶', 'روز۷'];
    const paths = [];
    const colors = ['#06D6D6', '#22C55E', '#FACC15', '#C084FC', '#38BDF8', '#FB923C', '#94A3B8', '#F87171', '#4ADE80', '#A855F7'];
    for (let p = 0; p < 10; p++) {
        const path = [base]; let cur = base;
        for (let d = 1; d <= 7; d++) { cur *= (1 + 0.001 + (Math.random()*2-1)*0.008); path.push(parseFloat(cur.toFixed(2))); }
        paths.push({ label: `سناریو ${p+1}`, data: path, borderColor: colors[p], borderWidth: 1.5, pointRadius: 0, tension: 0.2 });
    }
    if (window.multiCharts["monteCarloChart"]) window.multiCharts["monteCarloChart"].destroy();
    window.multiCharts["monteCarloChart"] = new Chart(ctx, {
        type: 'line',
        data: { labels: days, datasets: paths },
        options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { x: { ticks: { color: '#94A3B8' }, grid: { color: '#1E293B' } }, y: { ticks: { color: '#94A3B8' }, grid: { color: '#1E293B' } } } }
    });
}

function clean_json_value(x)
    if x isa AbstractFloat
        return isfinite(x) ? x : nothing
    elseif x isa AbstractString || x isa Symbol
        return String(x)
    elseif x isa Dates.TimeType
        return string(x)
    elseif x isa Dict
        return Dict(string(k) => clean_json_value(v) for (k, v) in x)
    elseif x isa AbstractVector
        return [clean_json_value(v) for v in x]
    elseif x isa Number || x isa Bool || x === nothing
        return x
    else
        return string(x)
    end
end

function safe_date_str(d)
    if d isa Dates.TimeType
        return Dates.format(d, "yyyy-mm-dd")
    elseif d isa AbstractString
        return length(d) >= 10 ? String(d)[1:10] : String(d)
    else
        return string(d)
    end
end

function serialize_sentiment(s::Union{NewsSentimentResult,Nothing})
    s === nothing && return Dict("available" => false)
    return Dict(
        "available"     => true,
        "overall_score" => s.overall_score,
        "gold_score"    => s.gold_score,
        "usd_score"     => s.usd_score,
        "iran_score"    => s.iran_score,
        "alert_level"   => s.alert_level,
        "summary"       => s.summary,
        "last_updated"  => Dates.format(s.last_updated, "yyyy-mm-dd HH:MM:SS"),
        "news_count"    => length(s.news_items),
        "top_news"      => [
            Dict(
                "title"    => item.title,
                "source"   => item.source,
                "score"    => item.score,
                "keywords" => item.keywords[1:min(4, end)]
            ) for item in s.news_items[1:min(12, end)]
        ]
    )
end

function serialize_relationships(rm)
    rm === nothing && return Dict("available" => false)
    relations = []
    for r in rm.relations[1:min(15, length(rm.relations))]
        rc = Dict{String,Any}()
        for (k,v) in r.regime_corr; rc[string(k)] = v; end
        push!(relations, Dict(
            "asset_b"       => r.asset_b,
            "pearson_20d"   => r.pearson_20d,
            "pearson_60d"   => r.pearson_60d,
            "lead_lag_days" => r.lead_lag_days,
            "lead_lag_corr" => r.lead_lag_corr,
            "beta_30d"      => r.beta_30d,
            "partial_corr"  => r.partial_corr,
            "direction"     => r.direction,
            "strength"      => r.strength,
            "regime_corr"   => rc,
            "dep_score"     => get(rm.dependency_scores, r.asset_b, 0.0)
        ))
    end
    sorted_deps = sort(collect(rm.dependency_scores), by=x->-x[2])
    top5 = [k for (k,v) in sorted_deps[1:min(5,length(sorted_deps))]]
    return Dict(
        "available"       => true,
        "computed_at"     => string(rm.computed_at),
        "relations"       => relations,
        "leading_assets"  => rm.leading_assets,
        "lagging_assets"  => rm.lagging_assets,
        "top_influencers" => top5
    )
end

function build_market_alert(data::Union{Dict,Nothing})
    data === nothing && return Dict("level" => "UNKNOWN", "message" => "--", "gold_move" => 0.0, "usd_move" => 0.0)
    try
        gp = data["gold_oz"]["prices"]
        up = data["usd_irr"]["prices"]
        gmove = length(gp) >= 2 ? abs(gp[end] - gp[end-1]) / gp[end-1] * 100.0 : 0.0
        umove = length(up) >= 2 ? abs(up[end] - up[end-1]) / up[end-1] * 100.0 : 0.0
        maxmove = max(gmove, umove)

        if maxmove > 2.0
            return Dict("level" => "HIGH", "message" => "بازار پرنوسان - پیش‌بینی با احتیاط", "gold_move" => round(gmove, digits=2), "usd_move" => round(umove, digits=2))
        elseif maxmove > 1.0
            return Dict("level" => "MEDIUM", "message" => "نوسان متوسط در بازار", "gold_move" => round(gmove, digits=2), "usd_move" => round(umove, digits=2))
        else
            return Dict("level" => "LOW", "message" => "نوسان بازار در بازه عادی و آرام", "gold_move" => round(gmove, digits=2), "usd_move" => round(umove, digits=2))
        end
    catch
        return Dict("level" => "UNKNOWN", "message" => "--", "gold_move" => 0.0, "usd_move" => 0.0)
    end
end

function update_data!()
    println("\n========== UPDATE START (HIGH-SPEED ENGINE) ==========")
    t_start = time()

    local_data = nothing
    local_analysis = nothing
    local_prediction = nothing
    local_backtest = nothing
    local_sentiment = nothing
    local_relationship_map = nothing
    local_portfolio_risk = nothing

    snap = get_state_snapshot()
    last_sentiment_cached = snap.last_sentiment
    last_backtest_cached = snap.last_backtest

    try
        println("  [1/7] fetch live market data (TGJU & Nobitex)...")
        local_data = DataFetcher.fetch_all_data(180)

        println("  [2/7] fetch macro data (Stooq DXY/US10Y/VIX)...")
        macrodata = MacroData.fetch_macro_data(180)
        local_data["macro"] = macrodata
        local_data["events"] = MacroData.fetch_upcoming_events(14)

        println("  [3/7] news sentiment analysis...")
        if last_sentiment_cached == DateTime(2000,1,1) || (now() - last_sentiment_cached) > Minute(30)
            try
                local_sentiment = fetch_news_sentiment(24)
                last_sentiment_cached = now()
            catch e
                local_sentiment = nothing
            end
        else
            local_sentiment = snap.sentiment
        end
        local_data["sentiment"] = local_sentiment

        println("  [4/7] cross-market correlations...")
        try
            others = Dict{String,Vector{Float64}}()
            others["usd"] = local_data["usd_irr"]["prices"]
            for c in local_data["currencies"]
                length(c.prices) >= 20 && (others[c.symbol] = c.prices)
            end
            macrod = get(local_data, "macro", Dict())
            if haskey(macrod, "dxy") && haskey(macrod["dxy"], "prices")
                others["DXY"] = macrod["dxy"]["prices"]
            end
            if haskey(macrod, "vix") && haskey(macrod["vix"], "prices")
                others["VIX"] = macrod["vix"]["prices"]
            end
            local_relationship_map = compute_relationships("gold", local_data["gold_oz"]["prices"], others; control_series = get(others, "usd", nothing))
        catch e
            local_relationship_map = nothing
        end

        println("  [5/7] quantitative forecasting (Gold, USD, Coin)...")
        local_analysis = calculate_all_ratios(local_data)
        local_prediction = predict_next_week(local_data, local_analysis)

        println("  [6/7] historical predictive backtest check...")
        if last_backtest_cached == DateTime(2000,1,1) || (now() - last_backtest_cached) > Hour(6)
            try
                local_backtest = run_historical_backtest(
                    local_data["gold_oz"]["prices"], local_data["gold_oz"]["dates"],
                    local_data["usd_irr"]["prices"], local_data["usd_irr"]["dates"],
                    local_data["iran_gold"]["prices"], local_data["iran_gold"]["dates"];
                    test_days=60, min_train=30
                )
                last_backtest_cached = now()
            catch e
                local_backtest = nothing
            end
        else
            local_backtest = snap.backtest
        end

        println("  [7/7] persisting records inside SQLite database...")
        try
            save_actual_snapshot!(local_data, local_analysis)
            save_prediction_run!(local_prediction)
        catch e
        end

        set_state_all!(
            local_data, local_analysis, local_prediction, local_backtest,
            local_sentiment, local_relationship_map,
            local_portfolio_risk, last_sentiment_cached, last_backtest_cached
        )

        println("========== HIGH-SPEED ASSEMBLY COMPLETED IN ($(round(time() - t_start, digits=2))s) ==========\n")
    catch e
        @error "CRITICAL CORE DATA COMPILATION FAILURE" exception=(e, catch_backtrace())
    end
end

function ensure_data()
    snap = get_state_snapshot()
    if snap.data === nothing
        println("📥 اولین راه‌اندازی: بارگذاری داده‌های اولیه به صورت همگام...")
        update_data!()
    end
end

function handle_index(req)
    try
        target_path = normpath(joinpath(TEMPLATES_DIR, "index.html"))
        if !isfile(target_path)
            return HTTP.Response(404, ["Content-Type" => "text/plain; charset=utf-8"], "قالب index.html یافت نشد.")
        end
        return HTTP.Response(
            200,
            [
                "Content-Type" => "text/html; charset=utf-8",
                "Cache-Control" => "no-store, no-cache, must-revalidate, max-age=0"
            ],
            read(target_path)
        )
    catch e
        return HTTP.Response(500, ["Content-Type" => "text/plain; charset=utf-8"], "Internal Server Error")
    end
end

function handle_static_file(raw_path::AbstractString)
    try
        path_str = String(split(raw_path, '?')[1])
        rel_file = replace(path_str, r"^/static/" => "")
        rel_file = replace(rel_file, r"^static/" => "")
        rel_file = replace(rel_file, "\\" => "/")
        
        target_path = normpath(joinpath(STATIC_DIR, rel_file))

        if !startswith(lowercase(target_path), lowercase(STATIC_DIR))
            return HTTP.Response(403, ["Content-Type" => "text/plain; charset=utf-8"], "Forbidden")
        end

        if !isfile(target_path)
            return HTTP.Response(404, ["Content-Type" => "text/plain; charset=utf-8"], "فایل استاتیک یافت نشد: $rel_file")
        end

        ext = lowercase(splitext(target_path)[2])
        ctype = get(Dict(
            ".css"   => "text/css; charset=utf-8",
            ".js"    => "application/javascript; charset=utf-8",
            ".json"  => "application/json; charset=utf-8",
            ".png"   => "image/png",
            ".jpg"   => "image/jpeg",
            ".svg"   => "image/svg+xml",
            ".ico"   => "image/x-icon"
        ), ext, "application/octet-stream")

        return HTTP.Response(
            200,
            [
                "Content-Type" => ctype,
                "Cache-Control" => "public, max-age=86400",
                "Access-Control-Allow-Origin" => "*"
            ],
            read(target_path)
        )
    catch e
        return HTTP.Response(500, ["Content-Type" => "text/plain; charset=utf-8"], "Internal Server Error")
    end
end

function handle_api(req)
    try
        ensure_data()
        resp = build_api_response()
        cleaned_resp = clean_json_value(resp)
        return HTTP.Response(
            200,
            [
                "Content-Type" => "application/json; charset=utf-8",
                "Access-Control-Allow-Origin" => "*",
                "Cache-Control" => "no-store, no-cache, must-revalidate, max-age=0"
            ],
            JSON3.write(cleaned_resp)
        )
    catch e
        @error "API response build error" exception=(e, catch_backtrace())
        return HTTP.Response(500, ["Content-Type" => "application/json; charset=utf-8"], JSON3.write(Dict("status" => "error", "message" => string(e))))
    end
end

function handle_refresh(req)
    try
        update_data!()
        return HTTP.Response(200, ["Content-Type" => "application/json; charset=utf-8"], JSON3.write(Dict("status" => "success", "message" => "داده‌ها به‌روزرسانی شدند.")))
    catch e
        return HTTP.Response(500, ["Content-Type" => "application/json; charset=utf-8"], JSON3.write(Dict("status" => "error", "message" => string(e))))
    end
end

function router(req::HTTP.Request)
    path = String(split(HTTP.URI(req.target).path, '?')[1])

    if path == "/favicon.ico"
        return HTTP.Response(204, ["Content-Type" => "image/x-icon"])
    elseif path == "/" || path == "/index.html"
        return handle_index(req)
    elseif startswith(path, "/static/")
        return handle_static_file(path)
    elseif path == "/api/data"
        return handle_api(req)
    elseif path == "/api/refresh"
        return handle_refresh(req)
    else
        return HTTP.Response(404, ["Content-Type" => "text/plain; charset=utf-8"], "صفحه مورد نظر یافت نشد.")
    end
end

    Storage.init_db()

    try

        while true
            sleep(60)
            try
                update_data!()
            catch e
            end
        end
    end

    HTTP.serve(router, "0.0.0.0", port)
end
